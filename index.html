function placeBid(amt){

  const p=curPlayer();if(!p||A.status!=='live')return;

  const tid=session.role==='admin'?selTeam:session.teamId;

  if(!tid)return notify('Select a bidding franchise first.','err');

  const t=teamOf(tid);if(!t)return;

  if(A.lead===tid)return notify(t.short+' already holds the top bid.','err');

  const floor=A.bid?ladderFor(p)[0].v:p.base;

  if(amt<floor)return notify('Bid must be at least '+fmt(floor)+'.','err');

  if(amt>t.budget)return notify(t.short+" can't afford "+fmt(amt)+'.','err');

  A.bid=amt;A.lead=tid;A.history.push({s:t.short,n:p.name,a:amt,t:Date.now()});

  saveA();sTick(amt);render();

}

}

function hammer(){

  if(!A.lead)return notify('No bid to hammer down!','err');

  const p=curPlayer();const t=teamOf(A.lead);if(!p||!t)return;

  if(t.budget<A.bid)return notify(t.short+' purse is below the winning bid — adjust the bag first.','err');

  t.budget-=A.bid;                       /* instant deduction */

  p.status='sold';p.soldTo=t.id;p.soldPrice=A.bid;   /* instant squad add */

  const purseLeft=t.budget;

  A.history.push({s:'SOLD',n:p.name,a:A.bid,t:Date.now()});

  const cap={name:p.name,img:p.img,cat:p.cat,team:t.name,short:t.short,color:t.color,price:A.bid,purseLeft:purseLeft};

  A={pid:null,bid:0,lead:null,status:'idle',history:A.history};

  selTeam=null;

  saveT();saveP();saveA();

  broadcast({type:'sold',cap:cap});

  sGavel();setTimeout(sSold,180);confetti();

  showSoldOverlay(cap,'sold');

}
